> `ast.ExceptHandler` 且 body 仅含 `Pass`。`logger._log` 与 `console()` 内部
> 因不能再调日志，改为写 `stderr` 留痕。

import ast, pathlib

hits = []
for p in pathlib.Path("src/guild_auto").rglob("*.py"):
    tree = ast.parse(p.read_text(encoding="utf-8"))
    for node in ast.walk(tree):
        if isinstance(node, ast.ExceptHandler):
            body = [n for n in node.body if not isinstance(getattr(n, "value", None), ast.Constant)]
            if len(body) == 1 and isinstance(body[0], ast.Pass):
                hits.append(f"{p}:{node.lineno}")
print("except:pass =", len(hits))  # 期望 0
